// Поля для обновления
$price_fields = ['price_low_1_3', 'price_low_4_8', 'price_low_9_15', 'price_low_16_30'];

$updated_count = 0;
$not_found = [];

echo "<h2>Обновление цен низкого сезона</h2>";

foreach ($prices_data as $car_title => $prices) {
    // Ищем пост по названию
    $posts = get_posts([
        'post_type' => 'rental_car',
        'title' => $car_title,
        'numberposts' => 1,
        'post_status' => 'publish'
    ]);
    
    if (empty($posts)) {
        // Пробуем найти по частичному совпадению
        $posts = get_posts([
            'post_type' => 'rental_car',
            's' => explode(' ', $car_title)[0], // Поиск по первому слову
            'numberposts' => 10,
            'post_status' => 'publish'
        ]);
        
        // Ищем наиболее похожий пост
        $best_match = null;
        foreach ($posts as $post) {
            if (strpos($post->post_title, explode(' ', $car_title)[0]) !== false) {
                $best_match = $post;
                break;
            }
        }
        
        if ($best_match) {
            $posts = [$best_match];
        }
    }
    
    if (!empty($posts)) {
        $post = $posts[0];
        echo "<p>Обновляем: <strong>{$post->post_title}</strong> (ID: {$post->ID})</p>";
        
        // Обновляем поля цен
        foreach ($price_fields as $index => $field) {
            update_field($field, $prices[$index], $post->ID);
        }
        
        $updated_count++;
    } else {
        $not_found[] = $car_title;
        echo "<p style='color: red;'>Не найден: <strong>{$car_title}</strong></p>";
    }
}

echo "<hr>";
echo "<h3>Результат:</h3>";
echo "<p>Обновлено автомобилей: <strong>{$updated_count}</strong></p>";

if (!empty($not_found)) {
    echo "<p>Не найдены следующие автомобили:</p>";
    echo "<ul>";
    foreach ($not_found as $car) {
        echo "<li>{$car}</li>";
    }
    echo "</ul>";
}

echo "<p><strong>Готово!</strong> Теперь вы можете проверить цены в админ панели WordPress.</p>";
